# Build Docker Image Locally

# Push Docker Image to Azure Container Registry
docker push mynewacrai2002026.azurecr.io/acr-demo-app:1.0

# Creating env variable in powershell

$env:PLAN_NAME = "asp-ai200-demo"
$env:WEBAPP_NAME = "demo-web-app-ai200"
$env:SKU = "B1"

#------Creating App Service PLAN
az appservice plan create `
  --name $env:PLAN_NAME `
  --resource-group $env:RG_NAME `
  --location westus2 `
  --sku $env:SKU `
  --is-linux

  #----Creating Web App
  az webapp create `
  --resource-group $env:RG_NAME `
  --plan $env:PLAN_NAME `
  --name $env:WEBAPP_NAME `
  --deployment-container-image-name "$env:ACR_NAME.azurecr.io/$env:IMAGE_NAME"
